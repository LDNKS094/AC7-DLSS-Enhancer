# All shaders are compiled here; C++ consumes only generated bytecode headers.
set(shader_output "${PROJECT_BINARY_DIR}/generated/shaders")
file(MAKE_DIRECTORY "${shader_output}")
function(ac7_shader target name source entry profile)
  set(header "${shader_output}/${name}.h")
  set(shader "${PROJECT_SOURCE_DIR}/shaders/${source}")
  add_custom_command(OUTPUT "${header}"
    COMMAND "${AC7_FXC_EXECUTABLE}" /nologo /Ges /Gis /O3 /T "${profile}" /E "${entry}"
      /Vn "${name}" /Fh "${header}" "${shader}"
    DEPENDS "${shader}" VERBATIM)
  target_sources(${target} PRIVATE "${header}")
  target_include_directories(${target} PRIVATE "${shader_output}")
endfunction()
