cmake_minimum_required(VERSION 3.20)
project(NrRequirementsQueries LANGUAGES CXX)
set(NGX_INCLUDE_DIR "" CACHE PATH "Directory with official nvsdk_ngx.h, nvsdk_ngx_defs.h and nvsdk_ngx_params.h")
set(NGX_LOADER_LIBRARY "" CACHE FILEPATH "Official x64 nvsdk_ngx_d.lib")
if(NOT EXISTS "${NGX_INCLUDE_DIR}/nvsdk_ngx_defs.h" OR NOT EXISTS "${NGX_INCLUDE_DIR}/nvsdk_ngx.h" OR NOT EXISTS "${NGX_INCLUDE_DIR}/nvsdk_ngx_defs_dlssnr.h" OR NOT EXISTS "${NGX_LOADER_LIBRARY}")
  message(FATAL_ERROR "Provide official NGX_INCLUDE_DIR (including NR defs) and NGX_LOADER_LIBRARY; SDK files are not vendored")
endif()
add_executable(query-nr-api query_requirements.cpp)
add_executable(query-nr-core query_core_requirements.cpp)
add_executable(preflight-nr preflight.cpp evaluate_frame.cpp evaluation_parameters.cpp ../../src/d3d11/shared_texture_bridge.cpp)
add_executable(gpu-chain gpu_chain.cpp evaluation_parameters.cpp ../../src/d3d11/shared_texture_bridge.cpp ../../src/d3d11/hdr_proxy.cpp ../../src/d3d11/guide_normalizer.cpp)
target_include_directories(gpu-chain PRIVATE ../../src/d3d11)
target_compile_definitions(gpu-chain PRIVATE "AC7_HDR_SHADER_PATH=L\"${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d11/hdr_proxy.hlsl\"")
target_link_libraries(gpu-chain PRIVATE d3d11 d3d12 d3dcompiler)
target_include_directories(preflight-nr PRIVATE ../../src/d3d11)
foreach(target query-nr-api query-nr-core preflight-nr gpu-chain)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(${target} PRIVATE "${NGX_INCLUDE_DIR}")
  target_link_libraries(${target} PRIVATE dxgi)
endforeach()
foreach(target query-nr-core preflight-nr gpu-chain)
  target_link_libraries(${target} PRIVATE "${NGX_LOADER_LIBRARY}" advapi32 shlwapi version wintrust crypt32 ole32)
endforeach()
target_link_libraries(preflight-nr PRIVATE d3d11 d3d12)
add_executable(hdr-reference hdr_reference.cpp)
target_compile_features(hdr-reference PRIVATE cxx_std_17)
target_compile_definitions(hdr-reference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
