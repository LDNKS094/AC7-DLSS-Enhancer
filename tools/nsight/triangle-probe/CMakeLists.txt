cmake_minimum_required(VERSION 3.20)
project(ac7_nsight_triangle_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This diagnostic probe is Windows-only")
endif()
add_executable(ac7-nsight-triangle-probe main.cpp)
target_compile_features(ac7-nsight-triangle-probe PRIVATE cxx_std_17)
target_compile_definitions(ac7-nsight-triangle-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ac7-nsight-triangle-probe PRIVATE d3d11 dxgi d3dcompiler user32)
