cmake_minimum_required(VERSION 3.20)
project(AC7NativeAdapter LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The current D3D11 validation module requires Windows")
endif()
add_library(ac7_guides STATIC src/d3d11/guide_normalizer.cpp)
target_compile_features(ac7_guides PUBLIC cxx_std_17)
target_compile_definitions(ac7_guides PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(ac7_guides PUBLIC src/d3d11)
target_link_libraries(ac7_guides PUBLIC d3d11 d3dcompiler)
add_executable(validate-guides tests/d3d11/validate_guides.cpp)
target_link_libraries(validate-guides PRIVATE ac7_guides)
add_library(ac7_hdr STATIC src/d3d11/hdr_proxy.cpp)
target_compile_features(ac7_hdr PUBLIC cxx_std_17)
target_compile_definitions(ac7_hdr PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_definitions(ac7_hdr PRIVATE "AC7_HDR_SHADER_PATH=L\"${CMAKE_CURRENT_SOURCE_DIR}/src/d3d11/hdr_proxy.hlsl\"")
target_include_directories(ac7_hdr PUBLIC src/d3d11)
target_link_libraries(ac7_hdr PUBLIC d3d11 d3dcompiler)
add_executable(validate-hdr-gpu tests/d3d11/validate_hdr.cpp)
target_link_libraries(validate-hdr-gpu PRIVATE ac7_hdr)
add_executable(gpu-live-pair tools/capture-addon/gpu_live_pair.cpp)
target_link_libraries(gpu-live-pair PRIVATE ac7_hdr ac7_guides)

add_library(ac7_sampler STATIC src/d3d11/two_frame_sampler.cpp)
target_compile_features(ac7_sampler PUBLIC cxx_std_17)
target_compile_definitions(ac7_sampler PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(ac7_sampler PUBLIC src/d3d11)
target_link_libraries(ac7_sampler PUBLIC d3d11 ole32)
add_executable(validate-two-frame-sampler tests/d3d11/validate_two_frame_sampler.cpp)
target_link_libraries(validate-two-frame-sampler PRIVATE ac7_sampler)

add_library(ac7_shared_bridge STATIC src/d3d11/shared_texture_bridge.cpp)
target_compile_features(ac7_shared_bridge PUBLIC cxx_std_17)
target_compile_definitions(ac7_shared_bridge PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(ac7_shared_bridge PUBLIC src/d3d11)
target_link_libraries(ac7_shared_bridge PUBLIC d3d11 d3d12 dxgi)
add_executable(validate-shared-bridge tests/d3d11/validate_shared_bridge.cpp)
target_link_libraries(validate-shared-bridge PRIVATE ac7_shared_bridge)

set(AC7_RESHADE_INCLUDE_DIR "" CACHE PATH "Official ReShade 6.8 public API20 headers for optional diagnostic addon")
if(AC7_RESHADE_INCLUDE_DIR)
  add_library(ac7-capture-addon MODULE tools/capture-addon/capture_addon.cpp)
  target_include_directories(ac7-capture-addon PRIVATE "${AC7_RESHADE_INCLUDE_DIR}")
  target_link_libraries(ac7-capture-addon PRIVATE ac7_sampler bcrypt)
  set_target_properties(ac7-capture-addon PROPERTIES PREFIX "" SUFFIX ".addon64")
endif()
