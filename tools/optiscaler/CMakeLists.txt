cmake_minimum_required(VERSION 3.20)
project(OptiscalerInterfaceProbe LANGUAGES CXX)
set(NGX_INCLUDE_DIR "" CACHE PATH "Official public NGX headers")
if(NOT EXISTS "${NGX_INCLUDE_DIR}/nvsdk_ngx.h")
  message(FATAL_ERROR "Provide NGX_INCLUDE_DIR; no SDK or OptiScaler binaries are vendored")
endif()
add_executable(optiscaler-interface probe.cpp ../nr/evaluation_parameters.cpp ../../src/d3d11/shared_texture_bridge.cpp)
target_compile_features(optiscaler-interface PRIVATE cxx_std_17)
target_compile_definitions(optiscaler-interface PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(optiscaler-interface PRIVATE "${NGX_INCLUDE_DIR}" ../../src/d3d11)
target_link_libraries(optiscaler-interface PRIVATE d3d11 d3d12 dxgi)

add_executable(color-interface color_probe.cpp ../nr/evaluation_parameters.cpp ../../src/d3d12/hdr_proxy.cpp ../../src/d3d11/hdr_proxy.cpp ../../src/d3d11/shared_texture_bridge.cpp)
target_compile_features(color-interface PRIVATE cxx_std_17)
target_include_directories(color-interface PRIVATE "${NGX_INCLUDE_DIR}")
target_compile_definitions(color-interface PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN "AC7_HDR_SHADER_PATH=L\"${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d11/hdr_proxy.hlsl\"")
target_link_libraries(color-interface PRIVATE d3d11 d3d12 dxgi d3dcompiler)

set(NGX_LOADER_LIBRARY "" CACHE FILEPATH "Official loader, needed only for the direct-route comparison")
if(EXISTS "${NGX_LOADER_LIBRARY}")
  add_executable(chain-direct-interface color_probe.cpp ../nr/evaluation_parameters.cpp ../../src/d3d12/hdr_proxy.cpp ../../src/d3d11/hdr_proxy.cpp ../../src/d3d11/shared_texture_bridge.cpp)
  target_compile_features(chain-direct-interface PRIVATE cxx_std_17)
  target_include_directories(chain-direct-interface PRIVATE "${NGX_INCLUDE_DIR}")
  target_compile_definitions(chain-direct-interface PRIVATE AC7_DIRECT_NGX NOMINMAX WIN32_LEAN_AND_MEAN "AC7_HDR_SHADER_PATH=L\"${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d11/hdr_proxy.hlsl\"")
  target_link_libraries(chain-direct-interface PRIVATE d3d11 d3d12 dxgi d3dcompiler "${NGX_LOADER_LIBRARY}" advapi32 shlwapi version wintrust crypt32 ole32)
endif()
